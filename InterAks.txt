public class Main {
    public static void main(String[] args) {
        System.out.println("Hello World");

        String str = "Applea Valley";

        // Define vowels (lowercase and uppercase)
        Set<Character> vowels = Set.of('a','e','i','o','u','A','E','I','O','U');

        // Split string into words and count vowels in each word
        // Map<String, Long> mp = 
        long coin = Stream.of(str.split("\\s"))
                .collect(Collectors.toMap(
                        word -> word, 
                        word -> word.chars() // IntStream of characters
                                .mapToObj(c -> (char)c) // convert int -> char
                                .filter(vowels::contains) // keep only vowels
                                .count()
                ))
                .values().stream().max(Long::compare).orElse(0L);
                
                System.out.println(coin);

        // System.out.println(mp);
    }
}
